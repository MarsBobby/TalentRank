/* ════ ANALYSIS ════ */
async function runAnalysis() {
  if (!S.fn || !S.files.length) return;
  const btn = document.getElementById('abtn');
  btn.disabled = true; btn.textContent = '评估中…';
  Object.values(S.charts).forEach(c => c.destroy()); S.charts = {}; S.fbs = {};
  document.getElementById('report-btn').classList.add('btn--hidden');
  showProg(true);

  const phases = [
    '数据资产盘点与字段分级', 'MNAR特征重构与纠偏', 'NLP语义提取与职级校准',
    '置信度评估与异常路由', 'AHP+EWM动态混合赋权', 'LambdaMART精排引擎执行',
    'Contextual Bandit初始化', 'SHAP解释性归因计算', '双轨制评估指标生成',
  ];
  let ps = 0;
  const tick = setInterval(() => {
    ps = Math.min(ps + 1, phases.length - 1);
    setProg(Math.round(ps / phases.length * 85), phases[ps]);
  }, 650);

  try {
    const fd = new FormData();
    S.files.forEach(f => fd.append('files', f));
    fd.append('function', S.fn);
    const r = await fetch('/api/analyze', { method:'POST', body:fd });
    clearInterval(tick);
    if (!r.ok) { const e = await r.json(); throw new Error(e.detail || 'HTTP ' + r.status); }
    const data = await r.json();
    S.result = data;
    setProg(100, '评估完成 · 正在加载结果...');
    await sleep(400);
    showProg(false);
    renderResults(data);
    toast('评估完成，共 ' + data.total + ' 名候选人', 'ok');
  } catch(e) {
    clearInterval(tick); showProg(false);
    toast('评估失败：' + e.message, 'err');
    btn.disabled = false; btn.textContent = '开始评估';
  }
}

/* ════ RENDER RESULTS ════ */
function renderResults(data) {
  // Stats strip
  const q = data.quality, dist = q.distribution || {};
  const ndcg = data.metrics['NDCG@5'] ?? data.metrics['NDCG@10'] ?? 0;
  const reviewCount = dist['red (<0.60)'] || 0;
  document.getElementById('s-total').innerHTML = `${data.total}<span class="stat-unit">人</span>`;
  document.getElementById('s-fn').textContent = data.function + ' 职能 · ' + data.run_time;

  const rvEl = document.getElementById('s-review');
  rvEl.innerHTML = `${reviewCount}<span class="stat-unit">人</span>`;
  rvEl.className = 'stat-val ' + (reviewCount > 0 ? 'stat-val--warn' : 'stat-val--ok');

  // Master list
  document.getElementById('master-hd-title').textContent = data.function + ' 职能';
  document.getElementById('master-hd-count').textContent = data.total + ' 人';
  const ml = document.getElementById('master-list');
  ml.innerHTML = '';
  data.candidates.forEach((c, i) => ml.appendChild(buildMasterItem(c, i)));

  collapseWizard();
  const ws = document.getElementById('workspace');
  ws.classList.add('show');

  // Show report button
  document.getElementById('report-btn').classList.remove('btn--hidden');

  // Auto-select first
  setTimeout(() => selectCandidate(0), 120);
}

/* ════ MASTER ITEM ════ */
function buildMasterItem(c, idx) {
  const el = document.createElement('div');
  const rankClass = c.rank <= 2 ? 'm-rank-' + c.rank : '';
  const cp = Math.round(c.confidence * 100);
  const cc = c.confidence >= .75 ? 'm-cf-g' : c.confidence >= .6 ? 'm-cf-a' : 'm-cf-r';
  if (c.needs_review) el.classList.add('flagged');
  el.className = ['master-item', c.needs_review ? 'flagged' : ''].filter(Boolean).join(' ');
  el.onclick = () => selectCandidate(idx);
  el.innerHTML = `
    <div class="m-rank ${rankClass}">#${c.rank}</div>
    <div class="m-info">
      <div class="m-name">${c.candidate_id}</div>
      <div class="m-meta">${c.target_position ? c.target_position.slice(0,14) + ' · ' : ''}${c.total_work_years ? c.total_work_years+'年' : 'L'+Math.round(c.s_level)}</div>
    </div>
    <div class="m-right">
      <div class="m-score">${c.score.toFixed(2)}</div>
      <div class="m-conf-strip"><div class="m-conf-fill ${cc}" style="width:${cp}%"></div></div>
    </div>`;
  return el;
}

/* ════ SELECT CANDIDATE ════ */
function selectCandidate(idx) {
  if (!S.result) return;
  S.selected = idx;
  const c = S.result.candidates[idx];

  // Update active state
  document.querySelectorAll('.master-item').forEach((el, i) => el.classList.toggle('active', i === idx));

  // Show detail
  document.getElementById('detail-empty').classList.add('hidden');
  const dc = document.getElementById('detail-content');
  dc.classList.remove('hidden');
  dc.classList.remove('entering');
  void dc.offsetWidth;
  dc.classList.add('entering');

  // Render
  renderDetail(c);
  switchTab('profile');

  // Draw radar after render
  if (S.charts.detail) { S.charts.detail.destroy(); delete S.charts.detail; }
  setTimeout(() => drawDetailRadar(c), 60);
}

/* ════ HR FEEDBACK ════ */
async function doFb(cid, action) {
  if (S.fbs[cid] === action) return;
  S.fbs[cid] = action;
  _applyFbButtonUI(cid, action);
  toast(`${cid} · ${ACT_LABEL[action]}`, 'ok');

  // If this is the currently shown candidate, update footer
  if (S.result && S.result.candidates[S.selected]?.candidate_id === cid) {
    const fbEl = document.getElementById('fb-' + cid);
    if (fbEl) fbEl.innerHTML = `<div class="hr-fb-status">${ICON.checksm} ${ACT_LABEL[action]}</div>`;
  }

  try {
    const resp = await fetch('/api/feedback', {
      method:'POST', headers:{'Content-Type':'application/json'},
      body: JSON.stringify({candidate_id:cid, action, function:S.fn})
    });
    if (resp.ok) {
      const data = await resp.json();
      const fbEl = document.getElementById('fb-' + cid);
      if (fbEl && data.reward !== undefined) {
        fbEl.innerHTML = `<div class="hr-fb-status">${ICON.checksm} ${ACT_LABEL[action]} · r=${data.reward}</div>`;
      }
    }
  } catch(e) { console.warn('反馈同步失败:', e.message); }
}

function _applyFbButtonUI(cid, action) {
  ['invite','bookmark','click','reject'].forEach(a => {
    const btn = document.getElementById(`hb-${cid}-${a}`);
    if (btn) btn.classList.remove('act-inv','act-bkm','act-vw','act-rej');
  });
  const activeBtn = document.getElementById(`hb-${cid}-${action}`);
  if (activeBtn) activeBtn.classList.add(ACT_CLASS[action]);
}

function scoreToPercentile(score, allScores) {
  const sorted=[...allScores].sort((a,b)=>b-a);
  const rank=sorted.indexOf(score)+1;
  return Math.round((1-rank/sorted.length)*100);
}
function getVerdict(rank,total,confidence) {
  if (rank===1 && confidence>=.6) return {cls:'v-str',txt:'强烈推荐'};
  if (rank<=Math.ceil(total*.33) && confidence>=.5) return {cls:'v-rec',txt:'推荐面试'};
  if (rank<=Math.ceil(total*.66)) return {cls:'v-obs',txt:'观望跟进'};
  return {cls:'v-no',txt:'暂不推荐'};
}
function getLevelDesc(sl) { return ({1:'基层执行(IC)',2:'初级管理层',3:'中级管理层',4:'总监级',5:'VP级',6:'C-Suite'})[Math.round(sl)]||'L'+Math.round(sl); }
function getConfidenceDesc(conf) { return conf>=.75?'信息完整，置信度高':conf>=.6?'部分字段推断，建议面试核实':'关键信息缺失较多，建议人工补全后重新评估'; }
function getMNARDesc(sm,km) { const p=[]; if(sm)p.push('管理规模未披露(MNAR建模处理)'); if(km)p.push('量化KPI未标注(成就语义信号替代)'); return p.length?p.join('；'):'核心量化指标均有数据支撑'; }
function getHRActionDesc(cid) { const a=S.fbs[cid]; if(!a) return null; return {action:a,desc:{invite:'已发起面试邀约',bookmark:'已收藏列入候选库',click:'已标记查看',reject:'已标记淘汰'}[a]||a}; }

    const groups={invite:[],bookmark:[],click:[],reject:[]};
    fbEntries.forEach(([cid,a])=>{ if(groups[a])groups[a].push(cid); });
    const gLbl={invite:'邀约面试',bookmark:'已收藏',click:'标记查看',reject:'淘汰'};
    Object.entries(groups).filter(([,v])=>v.length>0).forEach(([act,ids])=>{
      html+=`<div class="rpt-op-row">
        <span class="rpt-op-label ${act}">${gLbl[act]}</span>
        <span class="rpt-op-ids">${ids.join('、')}</span>
      </div>`;
    });
    html+=`</div>`;
  }

  const top1=data.candidates[0];
  const invList=Object.entries(S.fbs).filter(([,a])=>a==='invite').map(([id])=>id);
  html+=`<div class="rpt-section"><div class="rpt-section-title">综合建议</div>
    <div class="rpt-conclusion">
      <h4>系统综合建议</h4>
      <p>${top1?`首选：<strong>${top1.candidate_id}</strong>，综合得分 ${top1.score.toFixed(2)}，${getLevelDesc(top1.s_level)}，${top1.achievement_signal?'具备量化绩效记录，':''}建议优先安排面试。`:''}</p>
      <p class="rpt-p-mt">${invList.length?`HR已对 ${invList.join('、')} 发出邀约，建议面试中重点考察SHAP标识的弱势维度。`:'建议尽快对排名前三候选人启动初面，面试后通过系统反馈按钮更新决策。'}</p>
      <p class="rpt-conclusion-meta">生成时间：${now} · TalentRank · LambdaMART + LinUCB + SHAP</p>
    </div>
  </div>`;
  return html;
}
