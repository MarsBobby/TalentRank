  applyFilters() {
    if (!AppState.result) return;
    const term = document.getElementById('search-input').value.toLowerCase();
    const status = document.getElementById('filter-status').value;
    const sort = document.getElementById('sort-type').value;

    let list = [...AppState.result.candidates];
    
    // Filter
    list = list.filter(c => {
      const matchSearch = c.candidate_id.toLowerCase().includes(term) || (c.target_position||'').toLowerCase().includes(term);
      const dec = AppState.decisions[c.candidate_id]?.action;
      const matchStatus = status === 'all' ? true : status === 'none' ? !dec : dec === status;
      return matchSearch && matchStatus;
    });

    // Sort
    if (sort === 'rank') {
      list.sort((a,b) => b.score - a.score);
    } else if (sort === 'conf') {
      list.sort((a,b) => b.confidence - a.confidence);
    }

    this.renderMasterList(list);
    
    // Auto select first if exists
    if (list.length > 0 && !list.find(x => x.candidate_id === AppState.selectedId)) {
      this.selectCandidate(list[0].candidate_id);
    } else if (list.length === 0) {
      document.getElementById('detail-empty').style.display = 'flex';
      document.getElementById('detail-content').style.display = 'none';
    }
  },

  renderMasterList(list) {
    const container = document.getElementById('candidate-list');
    if (list.length === 0) {
      container.innerHTML = '<div style="padding:24px; text-align:center; color:var(--text-muted); font-size:13px;">无匹配候选人</div>';
      return;
    }

    container.innerHTML = list.map(c => {
      const dec = AppState.decisions[c.candidate_id]?.action;
      const decColor = dec === 'invite' ? 'var(--success)' : dec === 'reject' ? 'var(--danger)' : 'transparent';
      const decStyle = dec ? `border-left: 3px solid ${decColor};` : '';
      
      return `
      <div class="c-item ${AppState.selectedId === c.candidate_id ? 'active' : ''}" style="${decStyle}" onclick="App.selectCandidate('${c.candidate_id}')">
        <div class="c-item-rank">#${c.rank}</div>
        <div class="c-item-info">
          <div class="c-item-name">${c.candidate_id}</div>
          <div class="c-item-meta">${c.target_position || '职位待定'} · ${c.total_work_years}年经验</div>
          <div style="display:flex; gap:6px;">
            ${c.needs_review ? `<span class="tag tag-warn" style="font-size:10px; padding:2px 4px;">数据待核</span>` : ''}
            ${dec ? `<span class="tag tag-gray" style="font-size:10px; padding:2px 4px;">已决策</span>` : ''}
          </div>
        </div>
        <div class="c-item-score">${c.score.toFixed(2)}</div>
      </div>
    `}).join('');
  },

  selectCandidate(id) {
    AppState.selectedId = id;
    this.applyFilters(); // re-render list to show active state
    
    const c = AppState.result.candidates.find(x => x.candidate_id === id);
    if (!c) return;

    document.getElementById('detail-empty').style.display = 'none';
    document.getElementById('detail-content').style.display = 'flex';
    
    // Header
    document.getElementById('dh-score').textContent = c.score.toFixed(2);
    document.getElementById('dh-name').textContent = c.candidate_id;
    
    const tags = [];
    if(c.target_position) tags.push(`<span class="tag tag-blue">${c.target_position}</span>`);
    tags.push(`<span class="tag tag-gray">工作经验 ${c.total_work_years}年</span>`);
    if(c.achievement_signal) tags.push(`<span class="tag tag-blue" style="border-color:var(--success); color:var(--success); background:var(--success-bg);">量化绩效核实</span>`);
    document.getElementById('dh-tags').innerHTML = tags.join('');

    // Metrics Grid
    const lvlName = ['—','执行层','经理级','高级经理','总监级','副总裁','C-Suite'][Math.round(c.s_level)] || 'L'+c.s_level;
    document.getElementById('detail-metrics').innerHTML = `
      <div class="metric-box"><div class="mb-lbl">职级评估</div><div class="mb-val highlight">${lvlName}</div></div>
      <div class="metric-box"><div class="mb-lbl">战略主导项目</div><div class="mb-val">${c.major_projects_count} 个</div></div>
      <div class="metric-box"><div class="mb-lbl">团队管理规模</div><div class="mb-val">${c.span_missing ? '<span style="color:var(--text-muted); font-size:12px; font-weight:400;">数据未披露</span>' : c.span_of_control+' 人'}</div></div>
      <div class="metric-box"><div class="mb-lbl">从业经验</div><div class="mb-val">${c.total_work_years} 年</div></div>
    `;

    // SHAP NL Summary & Bars
    let summaryText = `该候选人主要优势在于<strong>${(c.shap_positive[0]?.cn_name||'综合能力')}</strong>`;
    if(c.shap_positive[1]) summaryText += `与<strong>${c.shap_positive[1].cn_name}</strong>`;
    if(c.shap_negative.length > 0) summaryText += `，但在<strong>${c.shap_negative[0].cn_name}</strong>方面存在潜在短板，建议面试重点考察。`;
    else summaryText += `，各项指标均衡且优于行业平均水平，属于高潜人才。`;
    document.getElementById('shap-summary').innerHTML = summaryText;

    let barsHtml = c.shap_positive.map(s => this._buildShapBar(s.cn_name, s.value, true)).join('');
    barsHtml += c.shap_negative.map(s => this._buildShapBar(s.cn_name, s.value, false)).join('');
    document.getElementById('shap-bars').innerHTML = barsHtml;

    // Diag (Data Completeness)
    const confPct = Math.round(c.confidence * 100);
    const confColor = confPct >= 75 ? 'var(--success)' : confPct >= 60 ? 'var(--warning)' : 'var(--danger)';
    document.getElementById('diag-content').innerHTML = `
      <div style="display:flex; align-items:center; gap:24px;">
        <div style="position:relative; width:80px; height:80px;">
          <svg viewBox="0 0 36 36" style="width:100%; height:100%; transform:rotate(-90deg);">
            <path d="M18 2.0845 a 15.9155 15.9155 0 0 1 0 31.831 a 15.9155 15.9155 0 0 1 0 -31.831" fill="none" stroke="var(--bg-active)" stroke-width="4"/>
            <path d="M18 2.0845 a 15.9155 15.9155 0 0 1 0 31.831 a 15.9155 15.9155 0 0 1 0 -31.831" fill="none" stroke="${confColor}" stroke-width="4" stroke-dasharray="${confPct}, 100"/>
          </svg>
          <div style="position:absolute; inset:0; display:flex; align-items:center; justify-content:center; font-family:var(--font-mono); font-weight:600; font-size:18px;">${confPct}</div>
        </div>
        <div style="flex:1;">
          <div style="font-weight:600; margin-bottom:8px;">简历特征完整度评估</div>
          <div style="font-size:13px; color:var(--text-body);">${c.needs_review ? '<span style="color:var(--warning);">发现关键信息缺失（如团队规模、历史薪酬等），系统已进行智能推断，建议在面试环节人工核对验证。</span>' : '关键评价维度数据完整，推荐结论置信度高。'}</div>
        </div>
      </div>
    `;

    // Render Radar
    this.renderRadar(c.radar);
    
    // Restore Actions
    const dec = AppState.decisions[id];
    document.querySelectorAll('.btn-action').forEach(b => b.classList.remove('active'));
    if(dec?.action) document.getElementById(`btn-act-${dec.action}`).classList.add('active');
    document.getElementById('action-note-input').value = dec?.note || '';
  },

  _buildShapBar(name, val, isPos) {
    const absVal = Math.abs(val);
    const pct = Math.min((absVal / 0.3) * 100, 100); // normalized against a max of ~0.3
    return `
      <div class="shap-bar-row">
        <div class="shap-lbl">${name}</div>
        <div class="shap-track"><div class="shap-fill ${isPos?'pos':'neg'}" style="width:${pct}%"></div></div>
        <div class="shap-val" style="color:${isPos?'var(--success)':'var(--danger)'}">${isPos?'+':''}${val.toFixed(2)}</div>
      </div>
    `;
  },

  renderRadar(candidateData) {
    const ctx = document.getElementById('radar-chart').getContext('2d');
    if (AppState.chartInstance) AppState.chartInstance.destroy();

    const labels = candidateData.map(d => d.label);
    const cData = candidateData.map(d => d.value);
    const industryData = [75, 70, 72, 68, 70, 75]; // Mock baseline

    AppState.chartInstance = new Chart(ctx, {
      type: 'radar',
      data: {
        labels: labels,
        datasets: [
          {
            label: '候选人表现',
            data: cData,
            backgroundColor: 'rgba(15, 82, 186, 0.15)',
            borderColor: '#0F52BA',
            borderWidth: 2,
            pointBackgroundColor: '#0F52BA',
            pointRadius: 4,
          },
          {
            label: '行业均值',
            data: industryData,
            backgroundColor: 'transparent',
            borderColor: '#94A3B8',
            borderWidth: 1.5,
            borderDash: [5, 5],
            pointRadius: 0,
          }
        ]
      },
      options: {
        responsive: true,
        maintainAspectRatio: false,
        scales: {
          r: {
            min: 0, max: 100,
            ticks: { display: false },
            grid: { color: 'rgba(0,0,0,0.05)' },
            pointLabels: { font: { family: "'Inter', sans-serif", size: 12 }, color: '#334155' }
          }
        },
        plugins: {
          legend: { position: 'bottom', labels: { boxWidth: 12, usePointStyle: true, font:{size:11} } },
          tooltip: {
            backgroundColor: 'rgba(15, 23, 42, 0.9)',
            titleFont: { size: 13 },
            bodyFont: { family: "'JetBrains Mono', monospace", size: 12 },
            padding: 10,
            callbacks: {
              label: (context) => ` ${context.dataset.label}: ${context.raw}分`
            }
          }
        }
      }
    });
  },

  // ════ HR ACTIONS ════
  takeAction(actionStr) {
    const cid = AppState.selectedId;
    if (!cid) return;

    if (actionStr === 'reject') {
      document.getElementById('cm-body').innerHTML = `确认将 <strong>${cid}</strong> 移入淘汰库？此操作将影响候选人后续评估权重。`;
      document.getElementById('confirm-modal').classList.add('active');
      document.getElementById('cm-confirm').onclick = () => {
        this._commitAction(cid, actionStr);
        this.closeModal();
      };
    } else {
      this._commitAction(cid, actionStr);
    }
  },

  _commitAction(cid, actionStr) {
    const prevDec = AppState.decisions[cid];
    AppState.undoStack = { cid, previous: prevDec ? {...prevDec} : null }; // save state for undo
    
    if(!AppState.decisions[cid]) AppState.decisions[cid] = {};
    AppState.decisions[cid].action = actionStr;
    
    this.applyFilters(); // refresh list UI
    this.selectCandidate(cid); // refresh detail UI
    
    const actionLabel = actionStr === 'invite' ? '已邀约' : actionStr === 'bookmark' ? '已收藏' : '已淘汰';
    this.toast(`候选人 ${cid} ${actionLabel}`, 'success', true);
  },

  undoAction() {
    if (!AppState.undoStack) return;
    const { cid, previous } = AppState.undoStack;
    if (previous) {
      AppState.decisions[cid] = previous;
    } else {
      delete AppState.decisions[cid];
    }
    AppState.undoStack = null;
    
    this.applyFilters();
    if(AppState.selectedId === cid) this.selectCandidate(cid);
    this.toast('操作已撤销', 'success');
  },

  saveNote(noteText) {
    const cid = AppState.selectedId;
    if (!cid) return;
    if(!AppState.decisions[cid]) AppState.decisions[cid] = {};
    AppState.decisions[cid].note = noteText;
    this.toast('决策备注已自动保存');
  },

  closeModal() {
    document.getElementById('confirm-modal').classList.remove('active');
  },

  // ════ UTILS ════
  toast(msg, type='success', withUndo=false) {
    const root = document.getElementById('toast-root');
    const el = document.createElement('div');
    el.className = `toast ${type}`;
    el.innerHTML = `
      <span>${msg}</span>
      ${withUndo ? `<span class="toast-action" onclick="App.undoAction(); this.parentElement.remove()">撤销</span>` : ''}
    `;
    root.appendChild(el);
    setTimeout(() => { el.style.opacity = '0'; }, 4000);
    setTimeout(() => el.remove(), 4300);
  },

  exportSingleReport() {
    this.toast('全量评估报告生成中...', 'success');
    // MOCK: Download simple CSV
    let csvContent = "data:text/csv;charset=utf-8,\uFEFF";
    csvContent += "姓名,评估得分,数据置信度,职级判定,从业年限,决策状态\n";
    const c = AppState.result.candidates.find(x => x.candidate_id === AppState.selectedId);
    const dec = AppState.decisions[c.candidate_id]?.action || '未决策';
    csvContent += `${c.candidate_id},${c.score},${c.confidence},L${c.s_level},${c.total_work_years},${dec}\n`;
    
    const encodedUri = encodeURI(csvContent);
    const link = document.createElement("a");
    link.setAttribute("href", encodedUri);
    link.setAttribute("download", `TalentRank_Report_${c.candidate_id}.csv`);
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
  },
  
  exportReport() {
    this.toast('批量报告导出中...', 'success');
    // Implement full table export similarly
  }
};
