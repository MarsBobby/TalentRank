  try{
    const fd=new FormData();

    fd.append('function',S.fn);
    const r=await fetch('/api/analyze',{method:'POST',body:fd});
    clearInterval(tick);
    if(!r.ok){const e=await r.json();throw new Error(e.detail||'HTTP '+r.status)}
    const data=await r.json();

/* ════ HR 反馈 — 重构版，修复无响应问题 ════ */
const ACT_CLASS={invite:'act-inv',bookmark:'act-bkm',click:'act-vw',reject:'act-rej'};
const ACT_LABEL={invite:'已发起邀约',bookmark:'已收藏',click:'已标记查看',reject:'已淘汰'};
const ACT_ICON={invite:'📨',bookmark:'⭐',click:'👁',reject:'✕'};

async function doFb(cid, action){
  // 防止重复点击同一动作
  if(S.fbs[cid]===action)return;

  // 立即更新 UI（乐观更新，不等待网络）
  _applyFbUI(cid, action);
  S.fbs[cid]=action;
  toast(`${cid} · ${ACT_LABEL[action]}`,'ok');

  // 异步发送到后端（如失败不影响前端展示）
  try{
    const resp=await fetch('/api/feedback',{
      method:'POST',
      headers:{'Content-Type':'application/json'},
      body:JSON.stringify({candidate_id:cid,action,function:S.fn})
    });
    if(resp.ok){
      const data=await resp.json();
      const reward=data.reward??'—';
      const kl=typeof data.kl_div==='number'?data.kl_div.toFixed(4):'—';
      const fbEl=document.getElementById('fb-'+cid);
      if(fbEl)fbEl.innerHTML=`<div class="fbconf">✓ ${ACT_LABEL[action]} · Bandit reward=${reward} · KL=${kl}</div>`;
    }
  }catch(e){
    console.warn('反馈后端同步失败（不影响操作记录）:',e.message);
  }
}

function _applyFbUI(cid, action){
  // 清除所有激活态
  ['invite','bookmark','click','reject'].forEach(a=>{
    const btn=document.getElementById(`hb-${cid}-${a}`);
    if(!btn)return;
    btn.classList.remove('act-inv','act-bkm','act-vw','act-rej','loading');
  });
  // 设置新激活态
  const activeBtn=document.getElementById(`hb-${cid}-${action}`);
  if(activeBtn)activeBtn.classList.add(ACT_CLASS[action]);
  // 立即显示结果
  const fbEl=document.getElementById('fb-'+cid);
  if(fbEl)fbEl.innerHTML=`<div class="fbconf">✓ ${ACT_LABEL[action]}</div>`;
}

/* ════════════════════════════════════════════════

function scoreToPercentile(score, allScores){
  const sorted=[...allScores].sort((a,b)=>b-a);
  const rank=sorted.indexOf(score)+1;
  const pct=Math.round((1-rank/sorted.length)*100);
  return pct;
}

function getVerdict(rank, total, confidence){
  if(rank===1&&confidence>=0.6) return {cls:'v-str',txt:'★ 强烈推荐'};
  if(rank<=Math.ceil(total*0.33)&&confidence>=0.5) return {cls:'v-rec',txt:'▲ 推荐面试'};
  if(rank<=Math.ceil(total*0.66)) return {cls:'v-obs',txt:'◎ 观望跟进'};
  return {cls:'v-no',txt:'✕ 暂不推荐'};
}

function getLevelDesc(sLevel){
  const n=Math.round(sLevel);
  const map={1:'基层执行层（IC），适合专员/助理级岗位',2:'初级管理层，具备项目主导能力',3:'中级管理层，可独立管理团队与预算',4:'高级管理层（总监级），具备跨部门协调与战略落地能力',5:'VP 级，具备业务单元负责人能力，适合汇报直线高管',6:'C-Suite 高管，具备公司级战略决策能力'};
  return map[n]||`职级 L${n}`;
}

function getConfidenceDesc(conf){
  if(conf>=0.75) return '简历信息完整，提取置信度高，数据可靠性强';
  if(conf>=0.6)  return '简历信息基本完整，部分字段通过推断获取，建议面试中核实';
  return '简历关键信息缺失较多，建议 HR 手动补全后重新评估';
}

function getMNARDesc(spanMissing, kpiMissing){
  const parts=[];
  if(spanMissing)parts.push('管理规模未明确披露（符合高阶人才简历精简惯例，已通过 MNAR 建模处理）');
  if(kpiMissing) parts.push('量化 KPI 未直接标注（已通过成就语义信号进行替代判断）');
  return parts.length?parts.join('；'):'核心量化指标均有数据支撑。';
}

function getHRActionDesc(cid){
  const action=S.fbs[cid];
  if(!action)return null;
  const map={invite:'HR 已发起面试邀约，建议优先安排初面',bookmark:'HR 已收藏，列入候选库持续跟进',click:'HR 已标记查看，需进一步了解后做决定',reject:'HR 已标记淘汰，本轮不纳入推进'};
  return {action,desc:map[action]||action,icon:ACT_ICON[action]};
}

      <p style="margin-top:8px;color:var(--t3);font-size:12px">报告生成时间：${now} · 系统版本：TalentRank v1.0 · 模型：LambdaMART + LinUCB + SHAP</p>
    </div>
  </div>`;

  return html;
}
